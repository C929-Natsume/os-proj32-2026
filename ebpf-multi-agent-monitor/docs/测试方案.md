# eBPF Multi-Agent Monitor — 完整测试方案

## 项目架构回顾

```
┌──────────────────────────────────────────────────────────────────────┐
│  L0: eBPF 内核探针     sys_behavior.bpf.c / ssl_sniffer.bpf.c        │
│  L1: 用户态 C 程序      loader / ringbuf_handler / redis_shipper     │
│  L2: C++ 分析引擎       trace_engine (语义上下文追踪)                 │
│  L3: Python 分析层      main_analyzer / audit_engine / intra/inter   │
│  L4: 基础设施           Redis / SQLite                               │
└──────────────────────────────────────────────────────────────────────┘
```

class TestSQLiteStore(unittest.TestCase):

        self.tmpdir = tempfile.mkdtemp()
        self.db_path = os.path.join(self.tmpdir, "test.db")
        self.store = None

    def tearDown(self):
        if self.store:
            self.store.close()
        import shutil
        shutil.rmtree(self.tmpdir, ignore_errors=True)

    def _create_store(self):
        # Patch path
        import sqlite_store
        original = sqlite_store.SQLiteStore._default_db_path
        sqlite_store.SQLiteStore._default_db_path = staticmethod(lambda: self.db_path)
        self.store = SQLiteStore()
        sqlite_store.SQLiteStore._default_db_path = original

    def test_db_creation(self):
        self._create_store()
        self.assertTrue(os.path.exists(self.db_path))

    def test_persist_event(self):
        self._create_store()
        self.store.persist_event({
            "type": 1, "pid": 100, "ppid": 0, "uid": 1000,
            "timestamp": time.time_ns(), "comm": "test",
            "agent_id": "agent_100", "main_pid": 100
        })
        import sqlite3
        conn = sqlite3.connect(self.db_path)
        count = conn.execute("SELECT COUNT(*) FROM events").fetchone()[0]
